idf_build_get_property(engine BLIP_WASM_ENGINE)
idf_build_get_property(deps BLIP_WASM_DEPS)
set(srcs "adapter.cpp")
set(includes "include")
if(NOT CMAKE_BUILD_EARLY_EXPANSION)
    if(engine STREQUAL "wamr")
        set(WAMR_ROOT_DIR "${deps}/wamr")
        set(expected "25bd7eb63e828e4bd242cc9b38d260b4b31c6605")
        set(checkout "${WAMR_ROOT_DIR}")
        set(WAMR_BUILD_PLATFORM "esp-idf")
        if(CONFIG_IDF_TARGET_ARCH_RISCV)
            set(WAMR_BUILD_TARGET "RISCV32_ILP32")
        else()
            set(WAMR_BUILD_TARGET "XTENSA")
        endif()
        set(WAMR_BUILD_INTERP 1)
        set(WAMR_BUILD_FAST_INTERP 1)
        set(WAMR_BUILD_AOT 0)
        set(WAMR_BUILD_JIT 0)
        set(WAMR_BUILD_FAST_JIT 0)
        set(WAMR_BUILD_LIBC_BUILTIN 0)
        set(WAMR_BUILD_LIBC_WASI 0)
        set(WAMR_BUILD_REF_TYPES 0)
        set(WAMR_BUILD_SIMD 0)
        set(WAMR_BUILD_BULK_MEMORY 1)
        set(WAMR_BUILD_INSTRUCTION_METERING 1)
        set(WAMR_DISABLE_HW_BOUND_CHECK 1)
        include("${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake")
        # This sandbox exposes no WASI, filesystem, sockets or executable code.
        # Their unused ESP-IDF port files also assume pre-IDF-6 APIs on C6.
        list(FILTER WAMR_RUNTIME_LIB_SOURCE EXCLUDE REGEX "/espidf_(file|socket|memmap|clock)\\.c$")
        list(APPEND srcs ${WAMR_RUNTIME_LIB_SOURCE})
        list(APPEND srcs "pool_memory.c")
        list(APPEND includes "${IWASM_DIR}/include" "${UTILS_SHARED_DIR}" "${PLATFORM_SHARED_DIR}" "${PLATFORM_SHARED_DIR}/../include" "${IWASM_COMMON_DIR}")
    elseif(engine STREQUAL "wasm3")
        set(checkout "${deps}/wasm3")
        set(expected "0cd38327f0c721e75172f4f1eeb55854dc0517af")
        foreach(name bind code compile core env exec function info module parse validate)
            list(APPEND srcs "${checkout}/source/m3_${name}.c")
        endforeach()
        list(APPEND includes "${checkout}/source")
    elseif(NOT engine STREQUAL "baseline")
        message(FATAL_ERROR "Unknown BLIP_WASM_ENGINE: ${engine}")
    endif()
    if(DEFINED checkout)
        execute_process(COMMAND git -C "${checkout}" rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
        if(NOT actual STREQUAL expected)
            message(FATAL_ERROR "Unexpected ${engine} revision: ${actual}")
        endif()
        execute_process(COMMAND git -C "${checkout}" status --porcelain OUTPUT_VARIABLE dirty COMMAND_ERROR_IS_FATAL ANY)
        if(NOT dirty STREQUAL "")
            message(FATAL_ERROR "Benchmark dependency must be unmodified")
        endif()
    endif()
endif()
idf_component_register(SRCS ${srcs} INCLUDE_DIRS ${includes} REQUIRES pthread lwip esp_timer heap)
target_compile_features(${COMPONENT_LIB} PRIVATE cxx_std_20)
set_target_properties(${COMPONENT_LIB} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES)
target_compile_definitions(${COMPONENT_LIB} PUBLIC BLIP_BENCH_ENGINE="${engine}")
if(engine STREQUAL "wamr")
    target_compile_definitions(${COMPONENT_LIB} PUBLIC BLIP_BENCH_WAMR=1 WASM_ENABLE_INTERP=1 WASM_ENABLE_INSTRUCTION_METERING=1)
elseif(engine STREQUAL "wasm3")
    target_compile_definitions(${COMPONENT_LIB} PUBLIC BLIP_BENCH_WASM3=1 d_m3MaxNativeStack=8192 d_m3MaxFunctionStackHeight=256 d_m3MaxLinearMemoryPages=1)
    if(CONFIG_IDF_TARGET_ARCH_XTENSA)
        # GCC 15 advertises musttail, but its Xtensa windowed ABI cannot emit it.
        target_compile_definitions(${COMPONENT_LIB} PUBLIC M3_HAS_TAIL_CALL=0)
    endif()
endif()
target_compile_options(${COMPONENT_LIB} PRIVATE -Wno-error -Wno-format)
