idf_build_get_property(deps BLIP_WASM_DEPS)
set(srcs "src/service.cpp" "src/module_upload.cpp" "src/esp_wasm_component.cpp" "src/wamr_runtime.cpp" "src/pool_memory.c")
set(engine_includes "")
if(NOT CMAKE_BUILD_EARLY_EXPANSION)
    idf_build_get_property(worker_enabled BLIP_ENABLE_WASM)
    if(worker_enabled AND CONFIG_IDF_TARGET_ESP32 AND
       (NOT CONFIG_FREERTOS_UNICORE OR NOT CONFIG_ESP32_IRAM_AS_8BIT_ACCESSIBLE_MEMORY))
        message(FATAL_ERROR "The ESP32 production WASM worker requires sdkconfig.wasm.esp32.defaults (single core and byte-accessible IRAM).")
    endif()
    if(NOT CONFIG_FREERTOS_HZ EQUAL 1000)
        message(FATAL_ERROR "BLIP WASM requires CONFIG_FREERTOS_HZ=1000. Apply sdkconfig.wasm.defaults in a fresh firmware build directory or update the existing sdkconfig.")
    endif()
    if(NOT deps)
        message(FATAL_ERROR "BLIP_WASM_DEPS must point to pinned runtime checkouts")
    endif()
    set(WAMR_ROOT_DIR "${deps}/wamr")
    execute_process(COMMAND git -C "${WAMR_ROOT_DIR}" rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT actual STREQUAL "25bd7eb63e828e4bd242cc9b38d260b4b31c6605")
        message(FATAL_ERROR "Unexpected WAMR revision: ${actual}")
    endif()
    execute_process(COMMAND git -C "${WAMR_ROOT_DIR}" status --porcelain OUTPUT_VARIABLE dirty COMMAND_ERROR_IS_FATAL ANY)
    if(NOT dirty STREQUAL "")
        message(FATAL_ERROR "WAMR dependency must be unmodified")
    endif()
    set(WAMR_BUILD_PLATFORM "esp-idf")
    if(CONFIG_IDF_TARGET_ARCH_RISCV)
        set(WAMR_BUILD_TARGET "RISCV32_ILP32")
    else()
        set(WAMR_BUILD_TARGET "XTENSA")
    endif()
    set(WAMR_BUILD_INTERP 1)
    set(WAMR_BUILD_FAST_INTERP 1)
    set(WAMR_BUILD_AOT 0)
    set(WAMR_BUILD_JIT 0)
    set(WAMR_BUILD_FAST_JIT 0)
    set(WAMR_BUILD_LIBC_BUILTIN 0)
    set(WAMR_BUILD_LIBC_WASI 0)
    set(WAMR_BUILD_LIB_PTHREAD 0)
    set(WAMR_BUILD_LIB_WASI_THREADS 0)
    set(WAMR_BUILD_SHARED_MEMORY 0)
    set(WAMR_BUILD_SHARED_HEAP 0)
    set(WAMR_BUILD_MULTI_MODULE 0)
    set(WAMR_BUILD_DUMP_CALL_STACK 0)
    # Host cancellation needs the synchronized suspend flag on loop backedges.
    # This internal manager exposes no guest threading imports.
    set(WAMR_BUILD_THREAD_MGR 1)
    set(WAMR_BUILD_REF_TYPES 0)
    set(WAMR_BUILD_SIMD 0)
    set(WAMR_BUILD_BULK_MEMORY 1)
    set(WAMR_BUILD_INSTRUCTION_METERING 1)
    set(WAMR_DISABLE_HW_BOUND_CHECK 1)
    include("${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake")
    list(FILTER WAMR_RUNTIME_LIB_SOURCE EXCLUDE REGEX "/espidf_(file|socket|memmap|clock)\\.c$")
    list(APPEND srcs ${WAMR_RUNTIME_LIB_SOURCE})
    list(APPEND engine_includes "${IWASM_DIR}/include" "${UTILS_SHARED_DIR}" "${PLATFORM_SHARED_DIR}" "${PLATFORM_SHARED_DIR}/../include" "${IWASM_COMMON_DIR}")
    # Keep the strict contract on BLIP sources; upstream retains its settings.
    set_source_files_properties(${WAMR_RUNTIME_LIB_SOURCE} PROPERTIES COMPILE_OPTIONS "-Wno-error;-Wno-format")
endif()
idf_component_register(SRCS ${srcs} INCLUDE_DIRS "include" PRIV_INCLUDE_DIRS ${engine_includes}
                       REQUIRES blip_core pthread lwip esp_timer heap freertos)
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/blip_component.cmake)
blip_configure_component(${COMPONENT_LIB})
